import re
import unicodedata


class PromptOptimizer:
    def __init__(self):
        self.animal_keywords = [
            "cavalo", "cavalos", "?gua", "?guas", "potro", "potros",
            "cachorro", "c?o", "c?es", "gato", "gatos", "le?o", "le?es",
            "tigre", "tigres", "on?a", "on?as", "elefante", "elefantes",
            "girafa", "girafas", "zebra", "zebras", "boi", "bois",
            "vaca", "vacas", "touro", "touros", "urso", "ursos",
            "lobo", "lobos", "macaco", "macacos", "p?ssaro", "p?ssaros",
            "animal", "animais",
        ]
        self.person_keywords = [
            "homem", "mulher", "menino", "menina", "crian?a",
            "pessoa", "pessoas", "jovem", "idoso", "idosa",
            "rapaz", "mo?a", "ciclista",
        ]
        self.vehicle_keywords = [
            "carro", "carros", "autom?vel", "ve?culo", "suv",
            "moto", "motocicleta", "caminh?o", "?nibus", "bicicleta",
            "bicicletas", "bike", "biciclista", "ciclista", "avi?o", "avi?es",
            "helic?ptero", "barco", "navio",
        ]
        self.landscape_keywords = [
            "paisagem", "praia", "montanha", "floresta", "campo",
            "cidade", "deserto", "lago", "rio", "cachoeira", "savana",
        ]
        self.object_keywords = [
            "mesa", "cadeira", "computador", "celular", "telefone",
            "rel?gio", "copo", "garrafa", "livro", "c?mera",
            "la?o", "lacinho", "gravata", "brinco",
        ]

    @staticmethod
    def _normalizar(texto: str) -> str:
        texto = unicodedata.normalize("NFKD", texto.lower())
        return "".join(
            c for c in texto if not unicodedata.combining(c)
        )

    def detectar_categoria(self, texto: str) -> str:
        normalizado = self._normalizar(texto)
        grupos = [
            ("animal", self.animal_keywords),
            ("pessoa", self.person_keywords),
            ("veiculo", self.vehicle_keywords),
            ("paisagem", self.landscape_keywords),
            ("objeto", self.object_keywords),
        ]
        categorias = [
            categoria for categoria, palavras in grupos
            if any(self._normalizar(p) in normalizado for p in palavras)
        ]
        return "+".join(categorias) if categorias else "geral"

    def otimizar(self, prompt: str) -> str:
        prompt = re.sub(r"\s+", " ", prompt.strip())
        if not prompt:
            return ""

        texto = self._normalizar(prompt)
        categoria = self.detectar_categoria(prompt)

        # Mant?m o pedido original e acrescenta apenas poucas instru??es.
        extras = []
        if "animal" in categoria:
            extras.extend(["realistic animal anatomy", "full body animal", "complete head and face", "wide shot", "entire body visible", "space around the subject"])
        if any(x in texto for x in ["cavalo", "cavalos", "egua", "eguas", "potro", "potros"]):
            extras.append("four visible legs and natural hooves")
        if "girafa" in texto or "girafas" in texto:
            extras.append("long neck and four visible legs")
            if "savana" in texto:
                extras.append("savanna landscape")
        if "pessoa" in categoria:
            extras.append("natural human anatomy")
        if "veiculo" in categoria:
            extras.append("complete vehicle visible")
        if any(x in texto for x in ["bicicleta", "bicicletas", "bike", "ciclista", "biciclista"]):
            extras.append("complete bicycle with both wheels visible")
        if "paisagem" in categoria:
            extras.append("natural perspective")
        if any(x in texto for x in ["lacinho", "laco"]):
            extras.append("clearly visible separate bow accessory")
        if "futurista" in texto:
            extras.append("futuristic")
        if "a noite" in texto:
            extras.append("night scene")
        if "por do sol" in texto:
            extras.append("sunset")
        if "nascer do sol" in texto:
            extras.append("sunrise")
        if "fotografia" in texto or "foto" in texto:
            extras.append("photorealistic")
        if "cinematografico" in texto or "cinematografica" in texto:
            extras.append("cinematic")

        # Remove complementos duplicados sem apagar a descri??o original.
        resultado = [prompt]
        vistos = {self._normalizar(prompt)}
        for extra in extras:
            chave = self._normalizar(extra)
            if chave not in vistos:
                resultado.append(extra)
                vistos.add(chave)

        # Limite conservador de palavras para reduzir risco de truncamento.
        palavras = " ".join(resultado).split()
        if len(palavras) > 45:
            palavras_originais = prompt.split()
            restantes = max(0, 45 - len(palavras_originais))
            return " ".join(palavras_originais + " ".join(resultado[1:]).split()[:restantes])

        return ", ".join(resultado)

import time

import torch
from diffusers import StableDiffusionPipeline, DPMSolverMultistepScheduler

from .config import (
    MODEL_PATH,
    OUTPUTS_DIR,
    DEVICE,
    IMAGE_WIDTH,
    IMAGE_HEIGHT,
    NUM_INFERENCE_STEPS,
    GUIDANCE_SCALE,
    SEED,
    NEGATIVE_PROMPT,
)

from .prompt_optimizer import PromptOptimizer


class GeraArteIA:

    def __init__(self, model_id: str = MODEL_PATH):
        self.model_id = model_id
        self.pipe = None
        self.prompt_optimizer = PromptOptimizer()

    def carregar_modelo(self):

        print("Carregando modelo local...")
        print(f"Modelo: {self.model_id}")
        print(f"Dispositivo: {DEVICE}")

        dtype = torch.float16 if DEVICE == "cuda" else torch.float32

        self.pipe = StableDiffusionPipeline.from_pretrained(
            self.model_id,
            torch_dtype=dtype,
            use_safetensors=True,
        )

        # Scheduler otimizado
        self.pipe.scheduler = DPMSolverMultistepScheduler.from_config(
            self.pipe.scheduler.config
        )

        if DEVICE == "cuda":
            # OtimizaÃ§Ãµes para a GTX 960 com 2 GB de VRAM
            self.pipe.enable_sequential_cpu_offload()
            self.pipe.enable_attention_slicing()

        print("Modelo carregado com sucesso.")
        print("Modo offline ativado.")

    def gerar(
        self,
        prompt: str,
        nome_arquivo: str = "imagem_gerada.png"
    ):

        if self.pipe is None:
            raise RuntimeError(
                "O modelo ainda nÃ£o foi carregado."
            )

        # OtimizaÃ§Ã£o automÃ¡tica do prompt
        prompt_original = prompt
        prompt_final = self.prompt_optimizer.otimizar(prompt)

        # O CLIP do Stable Diffusion 1.5 suporta no mÃ¡ximo 77 tokens.
        # O prompt Ã© limitado antes da inferÃªncia para evitar truncamento.
        tokens = self.pipe.tokenizer(
            prompt_final,
            truncation=True,
            max_length=77,
            add_special_tokens=True,
        )

        prompt_final = self.pipe.tokenizer.decode(
            tokens["input_ids"],
            skip_special_tokens=True,
            clean_up_tokenization_spaces=False,
        ).strip()

        categoria = self.prompt_optimizer.detectar_categoria(
            prompt_original
        )

        print()
        print("========================================")
        print("INICIANDO GERAÃ‡ÃƒO")
        print("========================================")
        print(f"Prompt original: {prompt_original}")
        print(f"Categoria detectada: {categoria}")
        print(f"Prompt otimizado: {prompt_final}")
        print(f"Steps: {NUM_INFERENCE_STEPS}")
        print(f"Guidance Scale: {GUIDANCE_SCALE}")
        print(f"Seed: {SEED}")

        inicio = time.perf_counter()

        generator = torch.Generator(
            device="cpu"
        ).manual_seed(SEED)

        resultado = self.pipe(
            prompt=prompt_final,
            negative_prompt=NEGATIVE_PROMPT,
            width=IMAGE_WIDTH,
            height=IMAGE_HEIGHT,
            num_inference_steps=NUM_INFERENCE_STEPS,
            guidance_scale=GUIDANCE_SCALE,
            generator=generator,
        )

        imagem = resultado.images[0]

        caminho = OUTPUTS_DIR / nome_arquivo

        imagem.save(caminho)

        tempo = time.perf_counter() - inicio

        print()
        print("========================================")
        print("IMAGEM GERADA COM SUCESSO!")
        print("========================================")
        print(f"Arquivo: {caminho}")
        print(f"Tempo: {tempo:.2f} segundos")

        if torch.cuda.is_available():

            memoria = (
                torch.cuda.max_memory_allocated()
                / 1024**3
            )

            print(f"Pico de VRAM: {memoria:.2f} GB")

            torch.cuda.reset_peak_memory_stats()

        return caminho

from datetime import datetime

from fastapi import FastAPI, HTTPException, Request
from fastapi.responses import FileResponse
from fastapi.staticfiles import StaticFiles
from fastapi.templating import Jinja2Templates
from pydantic import BaseModel

from app.config import MODEL_PATH, OUTPUTS_DIR
from app.generator import GeraArteIA


app = FastAPI(
    title="GeraArteIA",
    description="Sistema de geraÃ§Ã£o de imagens por inteligÃªncia artificial.",
    version="1.0.0",
)


app.mount(
    "/static",
    StaticFiles(directory="web/static"),
    name="static",
)


templates = Jinja2Templates(
    directory="web/templates"
)


class PedidoImagem(BaseModel):
    prompt: str


gerador = GeraArteIA(MODEL_PATH)

modelo_carregado = False


@app.get("/")
def inicio(request: Request):

    return templates.TemplateResponse(
        request=request,
        name="index.html",
    )


@app.get("/api/status")
def status():

    return {
        "sistema": "GeraArteIA",
        "modelo": MODEL_PATH,
        "modelo_carregado": modelo_carregado,
    }


@app.post("/api/gerar")
def gerar_imagem(pedido: PedidoImagem):

    global modelo_carregado

    prompt = pedido.prompt.strip()

    if not prompt:

        raise HTTPException(
            status_code=400,
            detail="Digite uma descriÃ§Ã£o para gerar a imagem.",
        )

    try:

        if not modelo_carregado:

            gerador.carregar_modelo()

            modelo_carregado = True

        data_hora = datetime.now().strftime(
            "%Y%m%d_%H%M%S"
        )

        nome_arquivo = f"imagem_{data_hora}.png"

        caminho = gerador.gerar(
            prompt=prompt,
            nome_arquivo=nome_arquivo,
        )

        return {
            "sucesso": True,
            "arquivo": caminho.name,
            "imagem": f"/api/imagem/{caminho.name}",
        }

    except Exception as erro:

        print(f"Erro na geraÃ§Ã£o: {erro}")

        raise HTTPException(
            status_code=500,
            detail=str(erro),
        )


@app.get("/api/imagem/{nome_arquivo}")
def imagem(nome_arquivo: str):

    caminho = OUTPUTS_DIR / nome_arquivo

    if not caminho.exists():

        raise HTTPException(
            status_code=404,
            detail="Imagem nÃ£o encontrada.",
        )

    return FileResponse(
        path=caminho,
        media_type="image/png",
        filename=nome_arquivo,
    )

        const campoPrompt = document.getElementById("prompt");

        function mostrarStatus(mensagem, tipo = "") {

        function atualizarHistorico() {

        async function gerarImagem() {
            const prompt = campoPrompt.value.trim();

            if (!prompt) {
                mostrarStatus("Digite uma descrição ou escolha um dos exemplos antes de gerar a imagem.", "erro");
                campoPrompt.focus();
                return;
            }

            botaoGerar.disabled = true;
            botaoGerar.textContent = "⏳ Gerando imagem...";
            mostrarStatus("Preparando o modelo e gerando sua imagem. Esse processo pode levar alguns minutos.");
            areaImagem.style.display = "none";

            try {
                const resposta = await fetch("/api/gerar", {
                    method: "POST",
                    headers: {
                        "Content-Type": "application/json",
                        "Accept": "application/json"
                    },
                    body: JSON.stringify({ prompt })
                });

                const tipo = resposta.headers.get("content-type") || "";
                const corpo = await resposta.text();
                let dados;

                if (tipo.includes("application/json")) {
                    try {
                        dados = JSON.parse(corpo);
                    } catch {
                        throw new Error("O servidor retornou uma resposta JSON inválida.");
                    }
                } else {
                    console.error("Resposta inesperada do servidor:", corpo);
                    throw new Error(
                        "O servidor não retornou JSON. Verifique se a aplicação FastAPI está funcionando."
                    );
                }

                if (!resposta.ok) {
                    throw new Error(dados.detail || dados.message || "Não foi possível gerar a imagem.");
                }

                if (!dados.imagem) {
                    throw new Error("A API não informou o endereço da imagem gerada.");
                }

                const urlImagem = dados.imagem + (dados.imagem.includes("?") ? "&" : "?") + "t=" + Date.now();
                mostrarImagem(urlImagem, prompt);

                imagensDaSessao.unshift({
                    url: urlImagem,
                    prompt,
                    arquivo: dados.arquivo || ""
                });
                atualizarHistorico();

                mostrarStatus("Imagem gerada com sucesso!", "sucesso");
            } catch (erro) {
                console.error("Erro no GeraArteIA:", erro);
                mostrarStatus("Erro: " + erro.message, "erro");
            } finally {
                botaoGerar.disabled = false;
                botaoGerar.innerHTML = "<span aria-hidden=\"true\">✦</span> Gerar imagem";
            }
        }

        botaoGerar.addEventListener("click", gerarImagem);

        document.getElementById("btnBaixar").addEventListener("click", () => {
            if (!imagemAtual) {
                mostrarStatus("Gere uma imagem antes de tentar baixar.", "erro");
                return;
            }

            const link = document.createElement("a");
            link.href = imagemAtual.url.split("?")[0];
            link.download = imagemAtual.arquivo || "geraarteia.png";
            document.body.appendChild(link);
            link.click();
            link.remove();
        });

        document.getElementById("btnLimpar").addEventListener("click", () => {
            imagemAtual = null;
            imagemEl.removeAttribute("src");
            areaImagem.style.display = "none";
            estadoVazio.style.display = "flex";
            campoPrompt.value = "";
            campoPrompt.dispatchEvent(new Event("input"));
            statusEl.style.display = "none";
        });

        campoPrompt.addEventListener("keydown", (evento) => {
            if ((evento.ctrlKey || evento.metaKey) && evento.key === "Enter") {
                gerarImagem();
            }
        });
